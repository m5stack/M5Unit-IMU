50template <class ReadReg>
51ImuKind detectBuiltinImu(ReadReg read)
52{
53 uint8_t id{};
54
55 // MPU family: WHO_AM_I at 0x68 register 0x75.
56 if (read(static_cast<uint8_t>(0x68), static_cast<uint8_t>(0x75), id)) {
57 switch (id) {
58 case 0x19:
59 return ImuKind::MPU6886;
60 case 0x68:
61 return ImuKind::MPU6050;
62 case 0x71:
63 return ImuKind::MPU9250;
64 default:
65 break; // not a known MPU id -> fall through to the BMI270 probe (BMI270 lives at 0x68 too)
66 }
67 }
68
69 // BMI270: CHIP_ID 0x24 at register 0x00, on either candidate address.
70 for (const uint8_t candidate : {static_cast<uint8_t>(0x68), static_cast<uint8_t>(0x69)}) {
71 if (read(candidate, static_cast<uint8_t>(0x00), id) && id == 0x24) {
72 return ImuKind::BMI270;
73 }
74 }
75
76 // SH200Q: WHO_AM_I 0x18 at 0x6C register 0x30.
77 if (read(static_cast<uint8_t>(0x6C), static_cast<uint8_t>(0x30), id) && id == 0x18) {
78 return ImuKind::SH200Q;
79 }
80
81 return ImuKind::None;
82}
