182 // AUX support, in the three stages start_periodic_measurement() drives them:
183 //
184 // setup mode -> detect (only while Unknown) -> data mode
185 //
186 // On success the interface is left in data mode, where the BMI270 auto-polls the BMM150 at 25Hz
187 // into REG_DATA_0..REG_DATA_7.
188
189 // Drives the three stages, skipping the detection once it has an answer. False means this unit
190 // runs as a plain 6-axis BMI270, which on most boards is the expected outcome, not a failure
191 bool setup_aux_bmm150();
192 // Switches the interface on and into manual (single-register) access, so the BMM150's own
193 // registers can be reached with aux_read_register8()/aux_write_register8()
194 bool aux_enter_setup_mode();
195 // Resets the BMM150, checks its chip ID and reads its factory trim into _mag_trim. Requires
196 // setup mode. Returning false is the normal outcome on a board that wires nothing here
197 bool aux_detect_bmm150();
198 // Applies the repetition, starts the BMM150's normal mode and hands the interface over to the
199 // BMI270's auto-poll. Requires setup mode and a preceding successful detection
200 bool aux_enter_data_mode();
201 // Polls REG_STATUS.aux_busy (bit2) until clear; gates every AUX manual transaction below
202 bool aux_wait_ready();
203 bool aux_write_register8(const uint8_t reg, const uint8_t val);
204 bool aux_read_register8(const uint8_t reg, uint8_t& out);
205 // Manual burst read of 8 bytes starting at a BMM150 register, via the AUX bridge
206 bool aux_read_burst8(const uint8_t bmm150_reg, std::array<uint8_t, 8>& out);
207 // Reads the BMM150 factory trim (registers 0x5D-0x71) through 3 overlapping 8 byte AUX
208 // bursts (0x5D/0x65/0x6A) and maps them the same way as UnitBMM150::read_trim()
209 bool aux_read_trim(bmm150::trim_t& trim);
