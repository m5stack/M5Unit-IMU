403 // Per-capability auto-calibration state (M5Unified IMU_Class::offset_point_t, ported to
404 // physical-unit floats -- see ahrs_controller.cpp for update_stillness()/calibrate()).
405 struct axis_offset_state_t {
406 float offset[3]{}; // Current offset estimate, subtracted from every sample
407 float prev[3]{}; // Previous physical sample, for stillness diffing
408 float avg[3]{}; // Exponential moving average of recent samples
409 float radius{}; // Expected magnitude of (sample - offset) at rest
410 float tolerance{}; // Tolerance band around radius before nudging the offset
411 float noise{}; // Per-sample movement below which stillness accumulates
412 uint8_t average_shifter{1}; // avg update rate: avg += diff / (1 << average_shifter)
413 uint8_t stillness{}; // 0 = moving, saturates toward 255 while stationary
414 uint8_t strength{}; // 0 = auto-calibration off, 1-255 = adjustment speed
415 // Radius adaptation: 0 keeps `radius` fixed (correct for accel, whose sphere is 1G
416 // everywhere, and for gyro, whose sphere is the origin). Non-zero makes `radius` track the
417 // observed magnitude at rate 1/(1 << radius_shift) per calibrate() call -- needed for the
418 // magnetometer, whose sphere is the local field strength and therefore varies by more than
419 // a factor of two across the planet (~25-65uT). Must stay much slower than the offset
420 // correction itself, so that it settles on the average over orientations rather than
421 // chasing each individual sample (which would cancel the correction entirely).
422 uint8_t radius_shift{};
423 bool radius_seeded{}; // First observed magnitude has been taken as the initial radius
424
425 uint8_t update_stillness(const float sample[3]);
426 void calibrate();
427 };
