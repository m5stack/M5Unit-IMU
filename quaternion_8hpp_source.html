134 static inline Quaternion fromEuler(const float roll_deg, const float pitch_deg, const float yaw_deg)
135 {

137 const float roll = roll_deg * deg2rad;

142 const float sr = std::sin(roll * 0.5f);
143 const float cp = std::cos(pitch * 0.5f);

146 const float sy = std::sin(yaw * 0.5f);

148 return Quaternion(cr * cp * cy + sr * sp * sy, sr * cp * cy - cr * sp * sy, cr * sp * cy + sr * cp * sy,

160 inline void toEuler(float& roll_deg, float& pitch_deg, float& yaw_deg) const
161 {
162 constexpr float rad2deg{180.0f / QUATERNION_PI};
163
164 const float sinr_cosp = 2.0f * (w * x + y * z);
165 const float cosr_cosp = 1.0f - 2.0f * (x * x + y * y);
166 const float roll = std::atan2(sinr_cosp, cosr_cosp);
167
168 float sinp = 2.0f * (w * y - z * x);
169 if (sinp > 1.0f) {
170 sinp = 1.0f;
171 } else if (sinp < -1.0f) {
172 sinp = -1.0f;
173 }
174 const float pitch = std::asin(sinp);
175
176 const float siny_cosp = 2.0f * (w * z + x * y);
177 const float cosy_cosp = 1.0f - 2.0f * (y * y + z * z);
178 const float yaw = std::atan2(siny_cosp, cosy_cosp);
179
180 roll_deg = roll * rad2deg;
181 pitch_deg = pitch * rad2deg;
182 yaw_deg = yaw * rad2deg;
183 }
