105inline const uint8_t* axis_permutation_indices(const AxisPermutation p)

123inline AxisPermutation permutation_from_indices(const uint8_t (&p)[3])
124{
125 const uint8_t code = static_cast<uint8_t>((p[0] << 4) | (p[1] << 2) | p[2]);
126 switch (code) {
127 case (0 << 4) | (1 << 2) | 2:
128 return AxisPermutation::XYZ;
129 case (0 << 4) | (2 << 2) | 1:
130 return AxisPermutation::XZY;
131 case (1 << 4) | (0 << 2) | 2:
132 return AxisPermutation::YXZ;
133 case (1 << 4) | (2 << 2) | 0:
134 return AxisPermutation::YZX;
135 case (2 << 4) | (0 << 2) | 1:
136 return AxisPermutation::ZXY;
137 case (2 << 4) | (1 << 2) | 0:
138 return AxisPermutation::ZYX;
139 default:
140 return AxisPermutation::XYZ;
141 }
142}

153inline AxisOrder compose(const AxisOrder& first, const AxisOrder& second)
154{
155 const uint8_t* pa = axis_permutation_indices(first.permutation);
156 const uint8_t* pb = axis_permutation_indices(second.permutation);
157 const bool inv_a[3] = {first.invert_x, first.invert_y, first.invert_z};
158 const bool inv_b[3] = {second.invert_x, second.invert_y, second.invert_z};
159
160 uint8_t pc[3]{};
161 bool inv_c[3]{};
162 for (int i = 0; i < 3; ++i) {
163 pc[i] = pa[pb[i]];
164 inv_c[i] = inv_b[i] != inv_a[pb[i]];
165 }
166 AxisOrder result;
167 result.permutation = permutation_from_indices(pc);
168 result.invert_x = inv_c[0];
169 result.invert_y = inv_c[1];
170 result.invert_z = inv_c[2];
171 return result;
172}
