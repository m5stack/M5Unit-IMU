55// Output rate in milli-Hz at OverSampling::X16, indexed by DataRate. Milli-Hz rather than Hz so the
56// oversampling division below stays exact enough at the slow end (0.63Hz / 128 is 4.9mHz)
57constexpr uint32_t OUTPUT_RATE_MHZ[8] = {80000, 40000, 20000, 10000, 5000, 2500, 1250, 630};
59
62constexpr uint32_t output_rate_mhz(const DataRate rate, const OverSampling os)
63{
64 return OUTPUT_RATE_MHZ[static_cast<uint8_t>(rate)] >> static_cast<uint8_t>(os);
65}
66
69constexpr uint32_t interval_ms(const DataRate rate, const OverSampling os)
70{
71 return (1000000U + output_rate_mhz(rate, os) - 1) / output_rate_mhz(rate, os);
72}

75constexpr uint8_t ctrl_reg1_active(const DataRate rate, const OverSampling os)
76{
77 return static_cast<uint8_t>((static_cast<uint8_t>(rate) << 5) | (static_cast<uint8_t>(os) << 3) | 0x01);

92struct Data {
93 std::array<uint8_t, 7> raw{};
94
96 inline bool drdy() const
97 {
98 return (raw[0] & 0x08) != 0;
99 }
101 inline int16_t x_raw() const
102 {
103 return static_cast<int16_t>((raw[1] << 8) | raw[2]);
104 }
106 inline int16_t y_raw() const
107 {
108 return static_cast<int16_t>((raw[3] << 8) | raw[4]);
109 }
111 inline int16_t z_raw() const
112 {
113 return static_cast<int16_t>((raw[5] << 8) | raw[6]);
114 }
115
117 inline int32_t adj_x() const
118 {
119 return rescale(x_raw());
120 }
122 inline int32_t adj_y() const
123 {
124 return rescale(y_raw());
125 }
127 inline int32_t adj_z() const
128 {
129 return rescale(z_raw());
130 }
131
132private:
133 static inline int32_t rescale(const int16_t raw_axis)
134 {
135 // Native 0.10µT/LSB -> library 16 LSB/µT: raw * 0.10 * 16
136 return static_cast<int32_t>(lroundf(raw_axis * RAW_UT_PER_LSB * OUTPUT_LSB_PER_UT));
137 }
138};
