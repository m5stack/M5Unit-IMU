183 inline int16_t comp_x() const
184 {
185 const int16_t data_x = x_raw();
186 if (data_x == internal::XY_OVERFLOW_RAW) {
187 return OVERFLOW_OUTPUT;
188 }
189 uint16_t base = rhall();
190 if (base == 0) {
191 base = trim.dig_xyz1;
192 }
193 if (base == 0) {
194 return OVERFLOW_OUTPUT;
195 }
196
197 const int32_t xyz1_scaled = static_cast<int32_t>(trim.dig_xyz1) * 16384;
198 const uint16_t normalized = static_cast<uint16_t>(xyz1_scaled / base) - static_cast<uint16_t>(0x4000);
199 const int16_t centered = static_cast<int16_t>(normalized);
200 const int32_t squared = static_cast<int32_t>(centered) * static_cast<int32_t>(centered);
201 const int32_t cross_term = static_cast<int32_t>(trim.dig_xy2) * (squared / 128);
202 const int32_t xy1_scaled = static_cast<int32_t>(static_cast<int16_t>(trim.dig_xy1) * 128);
203 const int32_t linear_term = static_cast<int32_t>(centered) * xy1_scaled;
204 const int32_t combined = ((cross_term + linear_term) / 512) + static_cast<int32_t>(0x100000);
205 const int32_t gain = static_cast<int32_t>(static_cast<int16_t>(trim.dig_x2) + static_cast<int16_t>(0xA0));
206 const int32_t scale = (combined * gain) / 4096;
207 const int32_t applied = static_cast<int32_t>(data_x) * scale;
208
209 int16_t result = static_cast<int16_t>(applied / 8192);
210 result = static_cast<int16_t>((result + static_cast<int16_t>(trim.dig_x1) * 8) / 16);
211 return result;
212 }
214 inline int16_t comp_y() const
215 {
216 const int16_t data_y = y_raw();
217 if (data_y == internal::XY_OVERFLOW_RAW) {
218 return OVERFLOW_OUTPUT;
219 }
220 uint16_t base = rhall();
221 if (base == 0) {
222 base = trim.dig_xyz1;
223 }
224 if (base == 0) {
225 return OVERFLOW_OUTPUT;
226 }
227
228 const int32_t normalized_full = (static_cast<int32_t>(trim.dig_xyz1) * 16384) / base;
229 const uint16_t normalized = static_cast<uint16_t>(normalized_full) - static_cast<uint16_t>(0x4000);
230 const int16_t centered = static_cast<int16_t>(normalized);
231 const int32_t squared = static_cast<int32_t>(centered) * static_cast<int32_t>(centered);
232 const int32_t cross_term = static_cast<int32_t>(trim.dig_xy2) * (squared / 128);
233 const int32_t xy1_scaled = static_cast<int32_t>(static_cast<int16_t>(trim.dig_xy1) * 128);
234 const int32_t combined_num = cross_term + static_cast<int32_t>(centered) * xy1_scaled;
235 const int32_t combined = combined_num / 512;
236 const int32_t gain = static_cast<int32_t>(static_cast<int16_t>(trim.dig_y2) + static_cast<int16_t>(0xA0));
237 const int32_t scale = ((combined + static_cast<int32_t>(0x100000)) * gain) / 4096;
238 const int32_t applied = static_cast<int32_t>(data_y) * scale;
239
240 int16_t result = static_cast<int16_t>(applied / 8192);
241 result = static_cast<int16_t>((result + static_cast<int16_t>(trim.dig_y1) * 8) / 16);
242 return result;
243 }
245 inline int16_t comp_z() const
246 {
247 const int16_t data_z = z_raw();
248 if (data_z == internal::Z_OVERFLOW_RAW) {
249 return OVERFLOW_OUTPUT;
250 }
251 const uint16_t rh = rhall();
252 if (trim.dig_z2 == 0 || trim.dig_z1 == 0 || rh == 0 || trim.dig_xyz1 == 0) {
253 return OVERFLOW_OUTPUT;
254 }
255
256 const int16_t hall_offset =
257 static_cast<int16_t>(static_cast<int16_t>(rh) - static_cast<int16_t>(trim.dig_xyz1));
258 const int32_t z3_term = (static_cast<int32_t>(trim.dig_z3) * static_cast<int32_t>(hall_offset)) / 4;
259 const int32_t z_delta = static_cast<int32_t>(data_z - trim.dig_z4) * 32768;
260 const int32_t z1_raw = static_cast<int32_t>(trim.dig_z1) * (static_cast<int16_t>(rh) * 2);
261 const int16_t z1_rounded = static_cast<int16_t>((z1_raw + 32768) / 65536);
262
263 int32_t result = (z_delta - z3_term) / (trim.dig_z2 + z1_rounded);
264
265 if (result > internal::Z_POSITIVE_SATURATION) {
266 result = internal::Z_POSITIVE_SATURATION;
267 } else if (result < internal::Z_NEGATIVE_SATURATION) {
268 result = internal::Z_NEGATIVE_SATURATION;
269 }
270 result /= 16;
271
272 return static_cast<int16_t>(result);
273 }
