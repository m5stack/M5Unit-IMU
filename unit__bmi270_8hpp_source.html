297// Register addresses (I2C addr 0x68 (SDO=GND) / 0x69 (SDO=VDDIO, M5 board default))
298constexpr uint8_t REG_DATA_0{0x04}; // AUX (mag) burst start: 20 bytes = mag(8)+accel(6)+gyro(6), see REG_DATA_8
299constexpr uint8_t REG_STATUS{0x03}; // bit2 = aux_man_op busy (named REG_STATUS, not STATUS: ESP-IDF's
300 // ets_sys.h has a global `typedef enum STATUS STATUS`)
301constexpr uint8_t REG_DATA_8{0x0C}; // 12 bytes: accel X/Y/Z, gyro X/Y/Z (little-endian)
302constexpr uint8_t REG_INT_STATUS_1{0x1D}; // bit7 acc_drdy, bit6 gyr_drdy, bit5 aux_drdy
303constexpr uint8_t REG_TEMPERATURE_0{0x22}; // 2 bytes, little-endian, 0x8000 = invalid
304constexpr uint8_t REG_INTERNAL_STATUS{0x21};
305constexpr uint8_t REG_ACC_CONF{0x40};
306constexpr uint8_t REG_ACC_RANGE{0x41};
307constexpr uint8_t REG_GYR_CONF{0x42};
308constexpr uint8_t REG_GYR_RANGE{0x43};
309constexpr uint8_t REG_AUX_DEV_ID{0x4B};
310// Poll rate and trigger-readout offset for the AUX target (see UnitBMI270AuxBMM150)
311constexpr uint8_t REG_AUX_CONF{0x44};
312constexpr uint8_t REG_AUX_IF_CONF{
313 0x4C}; // bit7 manual_en, bit6 fcu_write_en, bits[3:2] man_rd_burst, bits[1:0] aux_rd_burst
314constexpr uint8_t REG_AUX_RD_ADDR{0x4D}; // write triggers an AUX read transaction
315constexpr uint8_t REG_AUX_WR_ADDR{0x4E}; // write triggers an AUX write transaction (of REG_AUX_WR_DATA)
316constexpr uint8_t REG_AUX_WR_DATA{0x4F};
317constexpr uint8_t REG_INIT_CTRL{0x59};
318constexpr uint8_t REG_INIT_ADDR_0{0x5B}; // followed by INIT_ADDR_1 (0x5C)
319constexpr uint8_t REG_INIT_DATA{0x5E};
320constexpr uint8_t REG_IF_CONF{0x6B}; // bit5 aux_en
321constexpr uint8_t REG_PWR_CONF{0x7C}; // bit0 adv_power_save
322constexpr uint8_t REG_PWR_CTRL{0x7D}; // bit0 aux_en, bit1 gyr_en, bit2 acc_en, bit3 temp_en

354template <class ReadReg>
355bool detectAddress(ReadReg read, uint8_t& addr)
356{
357 for (const uint8_t candidate : {static_cast<uint8_t>(0x68), static_cast<uint8_t>(0x69)}) {
358 uint8_t id{};
359 if (read(candidate, command::REG_CHIP_ID, id) && id == CHIP_ID) {
360 addr = candidate;
361 return true;
362 }
363 }
364 return false;
365}
