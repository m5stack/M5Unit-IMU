51constexpr Averaging clamp_averaging(const DataRate rate, const Averaging avg)
52{
53 return (rate == DataRate::Rate400Hz) ? Averaging::None
54 : (rate == DataRate::Rate200Hz)
55 ? (static_cast<uint8_t>(avg) > static_cast<uint8_t>(Averaging::X2) ? Averaging::X2 : avg)
56 : (rate == DataRate::Rate100Hz)
57 ? (static_cast<uint8_t>(avg) > static_cast<uint8_t>(Averaging::X4) ? Averaging::X4 : avg)
58 : avg;
59}

102struct otp_compensation_t {
103 float t_sens{};
104 float t_offs{};
105 float t0{23.0f};
106 float offset[3]{};
107 float sens[3]{};
108 float tco[3]{};
109 float tcs[3]{};
110 float cross[4]{};
111
113 inline float compensate_temp(const float t_raw) const
114 {
115 return (1.0f + t_sens) * t_raw + t_offs;
116 }
117
120 void compensate_mag(float (&m)[3], const float t_comp) const
121 {
122 for (size_t i = 0; i < 3; ++i) {
123 m[i] *= 1.0f + sens[i];
124 m[i] += offset[i];
125 m[i] += tco[i] * (t_comp - t0);
126 m[i] /= 1.0f + tcs[i] * (t_comp - t0);
127 }
128
129 const float cross_xy = cross[0];
130 const float cross_yx = cross[1];
131 const float cross_zx = cross[2];
132 const float cross_zy = cross[3];
133 const float denom = 1.0f - cross_yx * cross_xy;
134
135 const float x = (m[0] - cross_xy * m[1]) / denom;
136 const float y = (m[1] - cross_yx * m[0]) / denom;
137 const float z_delta = m[0] * (cross_yx * cross_zy - cross_zx) - m[1] * (cross_zy - cross_xy * cross_zx);
138 const float z = m[2] + z_delta / denom;
139
140 m[0] = x;
141 m[1] = y;
142 m[2] = z;
143 }
144};

204private:
205 // Assembles a little-endian 24bit unsigned value from raw[idx..idx+2]
206 inline uint32_t assemble24(const size_t idx) const
207 {
208 return static_cast<uint32_t>(raw[idx]) | (static_cast<uint32_t>(raw[idx + 1]) << 8) |
209 (static_cast<uint32_t>(raw[idx + 2]) << 16);
210 }
211 // Primary (pre-OTP) temperature conversion (degC)
212 inline float primary_temp() const
213 {
214 return static_cast<float>(temp_raw24()) * temp_res_raw24() + TEMP_OFFSET;
215 }
216 // Primary conversion (uT) followed by full OTP compensation for all 3 axes
217 inline std::array<float, 3> compensated_mag() const
218 {
219 std::array<float, 3> m{static_cast<float>(x_raw()) * lsb_to_ut_xy(),
220 static_cast<float>(y_raw()) * lsb_to_ut_xy(),
221 static_cast<float>(z_raw()) * lsb_to_ut_z()};
222 float raw_arr[3] = {m[0], m[1], m[2]};
223 comp.compensate_mag(raw_arr, comp.compensate_temp(primary_temp()));
224 m[0] = raw_arr[0];
225 m[1] = raw_arr[1];
226 m[2] = raw_arr[2];
227 return m;
228 }
229};
