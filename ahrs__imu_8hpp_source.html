283 inline bool setAxisOrder(const axis_t axis0, const axis_t axis1, const axis_t axis2)
284 {
285 const uint8_t idx[3] = {
286 static_cast<uint8_t>(axis0 >> 1),
287 static_cast<uint8_t>(axis1 >> 1),
288 static_cast<uint8_t>(axis2 >> 1),
289 };
290 if (idx[0] > 2 || idx[1] > 2 || idx[2] > 2 || idx[0] == idx[1] || idx[1] == idx[2] || idx[0] == idx[2]) {
291 return false;
292 }
293 AxisOrder order;
294 order.permutation = permutation_from_indices(idx);
295 order.invert_x = (axis0 & 1) != 0;
296 order.invert_y = (axis1 & 1) != 0;
297 order.invert_z = (axis2 & 1) != 0;
298 writeUserAxisOrder(order);
299 return true;
300 }

309 inline bool setAxisOrderRightHanded(const axis_t axis0, const axis_t axis1)

322 inline bool setAxisOrderLeftHanded(const axis_t axis0, const axis_t axis1)

327private:
328 // Right-handed third axis for a given first two (ported from m5::IMU_Class getAxis2)
329 static inline axis_t getAxis2(const axis_t axis0, const axis_t axis1)
330 {
331 uint8_t axis2 = static_cast<uint8_t>(((1 << (axis0 >> 1)) | (1 << (axis1 >> 1))) ^ 0x07);
332 axis2 = static_cast<uint8_t>(axis2 & 0x06);
333 const uint8_t ax0 = axis2 ? static_cast<uint8_t>(axis2 - 2) : 4;
334 axis2 = static_cast<uint8_t>(axis2 + (((axis0 & 0x06) == ax0) == ((axis0 & 1) == (axis1 & 1))));
335 return static_cast<axis_t>(axis2);
336 }
