152// Sampling interval (ms) per GyroODR raw register value, ceil(1000/Hz), indexed 0x00..0x0D
153// (only 0x06..0x0D are valid GyroODR values; others are unused placeholders).
154// constexpr namespace-scope array: internal linkage per translation unit (no ODR issue).
155constexpr uint32_t GYRO_ODR_INTERVAL_MS[14] = {
156 0, 0, 0, 0, 0, 0, // 0x00..0x05 (unused)
157 40, // 0x06 = 25Hz
158 20, // 0x07 = 50Hz
159 10, // 0x08 = 100Hz
160 5, // 0x09 = 200Hz
161 3, // 0x0A = 400Hz
162 2, // 0x0B = 800Hz
163 1, // 0x0C = 1600Hz
164 1, // 0x0D = 3200Hz
165};
166
168constexpr uint32_t odr_interval_ms(const GyroODR odr)
169{
170 return GYRO_ODR_INTERVAL_MS[static_cast<uint8_t>(odr)];
171}
