142private:
143 inline int32_t adjust(const int16_t raw_axis, const uint8_t asa_axis) const
144 {
145 // Hadj = raw * (ASA + 128) / 256, then rescale to 16 LSB/µT. The sensitivity comes from the
146 // sample's own BITM bit, so a 14bit sample is scaled by 0.6µT/LSB even if the chip has been
147 // switched to 16bit since
148 const float ut = raw_axis * ((asa_axis + 128) / 256.0f) * (bit16() ? RAW_UT_PER_LSB : RAW_UT_PER_LSB_14BIT);
149 return static_cast<int32_t>(lroundf(ut * OUTPUT_LSB_PER_UT));
150 }
