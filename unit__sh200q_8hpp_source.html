222constexpr uint8_t REG_OUTPUT{0x00}; // burst: accel6 + gyro6 + temp2 (little-endian) = 14 bytes
223constexpr uint8_t REG_ACC_CONFIG{0x0E}; // [7] HPF bypass / [4:3] ODR / [0] filter enable
224constexpr uint8_t REG_GYRO_CONFIG{0x0F}; // [4] HPF bypass / [3:1] ODR / [0] filter enable
225constexpr uint8_t REG_GYRO_DLPF{0x11}; // low bits: DLPF setting (ODR-relative bandwidth)
226constexpr uint8_t REG_FIFO_CONFIG{0x12}; // [7:6] buffer mode (00 = bypass)
227constexpr uint8_t REG_ACC_RANGE{0x16}; // [1:0] full scale
228constexpr uint8_t REG_GYRO_RANGE{0x2B}; // [2:0] full scale
229constexpr uint8_t REG_CHIP_ID{0x30}; // = 0x18
230// Undocumented registers: the datasheet register map ends at 0x30. The init sequence in begin()
231// mirrors the original M5StickC library (SH200Q.cpp) verbatim -- no public documentation exists
232// for these, and skipping them leaves the ADC in a bad state.
233constexpr uint8_t REG_ADC_RESET{0xC2}; // bit2 toggled 1 -> 0
234constexpr uint8_t REG_INIT_D8{0xD8}; // bit7 toggled 1 -> 0
235constexpr uint8_t REG_INIT_78{0x78}; // written 0x61 -> 0x00
236constexpr uint8_t REG_SET1{0xBA}; // written 0xC0
237constexpr uint8_t REG_SET2{0xCA}; // bit4 toggled 1 -> 0 ("ADC reset" in the original library)
